

CREATE OR REPLACE FUNCTION fn_update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_set_experience_years()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.experience_since IS NOT NULL THEN
        NEW.experience_years := EXTRACT(YEAR FROM CURRENT_DATE)::integer - NEW.experience_since;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_recalc_reviews_percent()
RETURNS TRIGGER AS $$
DECLARE
    v_specialist_id INTEGER;
BEGIN
    IF TG_OP = 'DELETE' THEN
        v_specialist_id := OLD.specialist_id;
    ELSE
        v_specialist_id := NEW.specialist_id;
    END IF;

    UPDATE specialists
    SET positive_reviews_percent = (
        SELECT ROUND(SUM(CASE WHEN rating >= 4 THEN 1 ELSE 0 END)::numeric / NULLIF(COUNT(*), 0) * 100, 1)::integer
        FROM specialist_reviews
        WHERE specialist_id = v_specialist_id AND is_active = true
    )
    WHERE id = v_specialist_id;

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TABLE service_categories (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    parent_id INTEGER DEFAULT NULL,
    sort_order INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT true,
    FOREIGN KEY (parent_id) REFERENCES service_categories(id) ON DELETE SET NULL
);
CREATE INDEX idx_service_categories_parent ON service_categories(parent_id);
COMMENT ON TABLE service_categories IS 'Справочник категорий услуг';

CREATE TABLE price_sections (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    section_name VARCHAR(200) NOT NULL,
    min_price DECIMAL(12,2),
    sort_order INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_price_sections_active ON price_sections(is_active);
CREATE TRIGGER trg_price_sections_updated_at
BEFORE UPDATE ON price_sections FOR EACH ROW EXECUTE FUNCTION fn_update_updated_at();
COMMENT ON TABLE price_sections IS 'Разделы прайс-листа';

CREATE TABLE uploads (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    original_filename VARCHAR(255) NOT NULL,
    stored_filename VARCHAR(255) NOT NULL,
    file_path VARCHAR(500) NOT NULL,
    file_size INTEGER,
    mime_type VARCHAR(100),
    table_name VARCHAR(50),
    record_id INTEGER,
    uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_uploads_table_record ON uploads(table_name, record_id);
COMMENT ON TABLE uploads IS 'Реестр загруженных файлов';

    description TEXT,

    seasons_count INTEGER,

    meta_description VARCHAR(500),
    is_active BOOLEAN DEFAULT true,
    sort_order INTEGER DEFAULT 0,
    slug VARCHAR(200) UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_projects_active ON projects(is_active);
CREATE INDEX idx_projects_date ON projects(project_date);
CREATE INDEX idx_projects_slug ON projects(slug);
CREATE TRIGGER trg_projects_updated_at
BEFORE UPDATE ON projects FOR EACH ROW EXECUTE FUNCTION fn_update_updated_at();
COMMENT ON TABLE projects IS 'Таблица проектов клиники';

CREATE TABLE project_partners (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    project_id INTEGER NOT NULL,
    partner_name VARCHAR(255) NOT NULL,
    partner_description TEXT,
    partner_logo_path VARCHAR(500),
    partner_type VARCHAR(100),
    sort_order INTEGER DEFAULT 0,

    FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE

);
CREATE INDEX idx_project_stages_project ON project_stages(project_id);
CREATE INDEX idx_project_stages_number ON project_stages(stage_number);

CREATE OR REPLACE VIEW v_services_with_prices AS
SELECT
    s.id, s.category, s.subcategory, s.service_name, s.photo_path,
    sp.service_code, sp.mics_code, sp.duration_min,
    sp.price_category_1, sp.price_category_2, sp.price_category_3,
    sp.price_single, sp.price_note
FROM services s
LEFT JOIN service_prices sp ON s.service_name = sp.service_name
WHERE s.is_active = true
ORDER BY s.category, s.subcategory, s.sort_order;

CREATE OR REPLACE VIEW v_active_promotions AS
SELECT id, title, category, description, discount_percent, photo_path, start_date, end_date
FROM promotions
WHERE is_active = true AND start_date <= CURRENT_DATE AND (end_date IS NULL OR end_date >= CURRENT_DATE)
ORDER BY start_date DESC;

CREATE OR REPLACE VIEW v_price_list AS
SELECT
    section_name, section_min_price, service_code, mics_code, service_name, category, duration_min,
    COALESCE(price_single::text,
        CONCAT(COALESCE(price_category_1, 0), ' / ', COALESCE(price_category_2, 0), ' / ', COALESCE(price_category_3, 0))
    ) AS price_display,
    price_note
FROM service_prices
WHERE is_active = true
ORDER BY section_name, sort_order, mics_code;

CREATE OR REPLACE VIEW v_latest_news AS
SELECT id, title, subtitle, category, photo_path, news_date, views_count
FROM news
WHERE is_active = true
ORDER BY news_date DESC, sort_order DESC
LIMIT 10;

-- Детальная страница проекта (ИСПРАВЛЕНО — ORDER BY во вложенном подзапросе)
CREATE OR REPLACE VIEW v_project_detail AS
SELECT
    p.id, p.title, p.description, p.full_description, p.context_description, p.project_description,
    p.photo_path, p.project_date, p.meta_title, p.meta_description,
    p.seasons_count, p.participants_count, p.experts_count, p.winners_count,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT pp.partner_name AS name, pp.partner_description AS description,
                pp.partner_logo_path AS logo, pp.partner_type AS type
         FROM project_partners pp
         WHERE pp.project_id = p.id AND pp.is_active = true
         ORDER BY pp.sort_order
     ) t) AS partners,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT ps.stage_number AS number, ps.stage_name AS name, ps.stage_description AS description
         FROM project_stages ps
         WHERE ps.project_id = p.id AND ps.is_active = true
         ORDER BY ps.sort_order
     ) t) AS stages,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT pg.photo_path AS path, pg.photo_alt AS alt,
                pg.photo_caption AS caption, pg.photo_type AS type
         FROM project_gallery pg
         WHERE pg.project_id = p.id AND pg.is_active = true
         ORDER BY pg.sort_order
     ) t) AS gallery,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT pt.full_name AS name, pt.role AS role,
                pt.role_category AS category, pt.photo_path AS photo
         FROM project_team pt
         WHERE pt.project_id = p.id AND pt.is_active = true
         ORDER BY pt.sort_order
     ) t) AS team
FROM projects p WHERE p.is_active = true;

CREATE OR REPLACE VIEW v_projects_stats AS
SELECT
    p.id, p.title, p.seasons_count, p.participants_count, p.experts_count, p.winners_count,
    (SELECT COUNT(*) FROM project_partners pp WHERE pp.project_id = p.id AND pp.is_active = true) AS partners_count,
    (SELECT COUNT(*) FROM project_stages ps WHERE ps.project_id = p.id AND ps.is_active = true) AS stages_count,
    (SELECT COUNT(*) FROM project_gallery pg WHERE pg.project_id = p.id AND pg.is_active = true) AS photos_count,
    (SELECT COUNT(*) FROM project_team pt WHERE pt.project_id = p.id AND pt.is_active = true) AS team_count
FROM projects p WHERE p.is_active = true ORDER BY p.project_date DESC;

     FROM (
         SELECT sr.patient_name AS patient, sr.review_text AS text,
                sr.rating AS rating, sr.review_date AS date
         FROM specialist_reviews sr
         WHERE sr.specialist_id = s.id AND sr.is_active = true
         ORDER BY sr.review_date DESC
     ) t) AS reviews,
    (SELECT COUNT(*) FROM specialist_reviews sr WHERE sr.specialist_id = s.id AND sr.is_active = true) AS reviews_count,
    (SELECT AVG(sr.rating) FROM specialist_reviews sr WHERE sr.specialist_id = s.id AND sr.is_active = true) AS avg_rating,
    (SELECT COUNT(*) FROM specialist_directions sd WHERE sd.specialist_id = s.id AND sd.is_active = true) AS directions_count,
    (SELECT COUNT(*) FROM specialist_certifications sc WHERE sc.specialist_id = s.id AND sc.is_active = true) AS certifications_count,
    (SELECT MIN(sc.expiry_date) FROM specialist_certifications sc WHERE sc.specialist_id = s.id AND sc.is_active = true) AS next_cert_expiry
FROM specialists s WHERE s.is_active = true ORDER BY s.sort_order, s.last_name;

CREATE OR REPLACE VIEW v_specialists_cards AS
SELECT
    s.id, s.full_name, s.position, s.short_description, s.photo_path, s.photo_alt, s.slug,
    s.experience_since, s.experience_years,
    s.operations_count, s.positive_reviews_percent, s.is_featured,
    (SELECT AVG(sr.rating) FROM specialist_reviews sr WHERE sr.specialist_id = s.id AND sr.is_active = true) AS avg_rating,
    (SELECT COUNT(*) FROM specialist_reviews sr WHERE sr.specialist_id = s.id AND sr.is_active = true) AS reviews_count
FROM specialists s WHERE s.is_active = true ORDER BY s.is_featured DESC, s.sort_order, s.last_name;

CREATE OR REPLACE VIEW v_specialists_expiring_certs AS
SELECT
    s.id, s.full_name, s.position, sc.specialty, sc.expiry_date, sc.note,
    (sc.expiry_date - CURRENT_DATE)::integer AS days_until_expiry
FROM specialists s
JOIN specialist_certifications sc ON s.id = sc.specialist_id
WHERE s.is_active = true AND sc.is_active = true AND sc.expiry_date <= CURRENT_DATE + INTERVAL '180 days'
ORDER BY sc.expiry_date ASC;

CREATE OR REPLACE VIEW v_specialists_rating AS
SELECT
    s.id, s.full_name, s.position, s.photo_path, s.slug,
    COUNT(sr.id) AS reviews_count,
    ROUND(AVG(sr.rating)::numeric, 2) AS avg_rating,
    SUM(CASE WHEN sr.rating = 5 THEN 1 ELSE 0 END) AS five_star_count,
    SUM(CASE WHEN sr.rating >= 4 THEN 1 ELSE 0 END) AS positive_count,
    ROUND((SUM(CASE WHEN sr.rating >= 4 THEN 1 ELSE 0 END)::numeric / NULLIF(COUNT(sr.id), 0) * 100), 1) AS positive_percent
FROM specialists s
LEFT JOIN specialist_reviews sr ON s.id = sr.specialist_id AND sr.is_active = true
WHERE s.is_active = true
GROUP BY s.id, s.full_name, s.position, s.photo_path, s.slug
HAVING COUNT(sr.id) > 0
ORDER BY avg_rating DESC, reviews_count DESC;

-- Команда проекта с группировкой по категориям ролей (ИСПРАВЛЕНО)
CREATE OR REPLACE VIEW v_project_team_by_role AS
SELECT
    p.id AS project_id,
    p.title AS project_title,
    pt.role_category,
    JSON_AGG(row_to_json(sub_t)) AS members
FROM projects p
JOIN (
    SELECT pt2.project_id, pt2.role_category, pt2.full_name, pt2.role, pt2.photo_path
    FROM project_team pt2
    WHERE pt2.is_active = true
    ORDER BY pt2.sort_order
) pt ON p.id = pt.project_id,
LATERAL (SELECT pt.full_name, pt.role, pt.photo_path) sub_t
WHERE p.is_active = true
GROUP BY p.id, p.title, pt.role_category
ORDER BY p.sort_order, pt.role_category;
