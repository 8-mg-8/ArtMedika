--   1. Услуги (разделы → подразделы → конкретные услуги + дочерние таблицы)

-- ============================================================
-- ФУНКЦИИ
-- ============================================================

CREATE OR REPLACE FUNCTION fn_update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_set_experience_years()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.experience_since IS NOT NULL THEN
        NEW.experience_years := EXTRACT(YEAR FROM CURRENT_DATE)::integer - NEW.experience_since;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ПРЕДСТАВЛЕНИЯ (VIEWS)
-- ============================================================

-- Дерево услуг: раздел → подраздел → услуга
CREATE OR REPLACE VIEW v_services_tree AS
SELECT
    sec.id AS section_id, sec.name AS section_name, sec.sort_order AS section_sort_order,
    sub.id AS subsection_id, sub.name AS subsection_name, sub.photo_path AS subsection_photo_path,
    s.id AS service_id, s.name AS service_name, s.slug, s.description, s.preparation, s.execution,
    s.sort_order AS service_sort_order
FROM service_sections sec
JOIN service_subsections sub ON sub.section_id = sec.id
JOIN services s ON s.subsection_id = sub.id
WHERE sec.is_active = true AND sub.is_active = true AND s.is_active = true
ORDER BY sec.sort_order, sec.id, sub.sort_order, sub.id, s.sort_order, s.id;

-- Услуги вместе с ценами
CREATE OR REPLACE VIEW v_services_with_prices AS
SELECT
    s.id, s.name, s.slug, s.description, sub.photo_path,
    sp.service_code, sp.mics_code, sp.duration_min,
    sp.price_category_1, sp.price_category_2, sp.price_category_3,
    sp.price_single
FROM services s
JOIN service_subsections sub ON sub.id = s.subsection_id
LEFT JOIN service_prices sp ON s.name = sp.service_name
WHERE s.is_active = true
ORDER BY s.sort_order, s.id;

-- Активные акции
CREATE OR REPLACE VIEW v_active_promotions AS
SELECT id, title, short_description, full_description, promotion_date, photo_path, slug
FROM promotions
WHERE is_active = true
  AND (promotion_date IS NULL OR promotion_date <= CURRENT_DATE)
ORDER BY promotion_date DESC, sort_order ASC;

-- Прайс-лист
CREATE OR REPLACE VIEW v_price_list AS
SELECT
    pl.name AS price_name,
    pl.name AS section_name,
    sp.service_code, sp.mics_code, sp.service_name, sp.category, sp.duration_min,
    COALESCE(sp.price_single::text,
        CONCAT(COALESCE(sp.price_category_1, 0), ' / ', COALESCE(sp.price_category_2, 0), ' / ', COALESCE(sp.price_category_3, 0))
    ) AS price_display
FROM service_prices sp
JOIN price_lists pl ON pl.id = sp.price_id
WHERE sp.is_active = true AND pl.is_active = true
ORDER BY pl.sort_order, pl.id, sp.sort_order, sp.mics_code;

-- Последние новости
CREATE OR REPLACE VIEW v_latest_news AS

FROM news
WHERE is_active = true
ORDER BY news_date DESC, sort_order DESC
LIMIT 10;

-- Детальная страница проекта
CREATE OR REPLACE VIEW v_project_detail AS
SELECT
    p.id, p.title, p.slug, p.description, p.project_date,
    p.photo_path, p.photo_alt,
    p.photo_2_path, p.photo_2_alt,
    p.photo_3_path, p.photo_3_alt,
    p.seasons_count, p.participants_count, p.experts_count, p.partners_count,
    pd.context_description, pd.project_description,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT ps.title AS title, ps.description AS description
         FROM project_seasons ps
         WHERE ps.project_id = p.id
         ORDER BY ps.sort_order
     ) t) AS seasons,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT pp.title AS title, pp.description AS description
         FROM project_participants pp
         WHERE pp.project_id = p.id
         ORDER BY pp.sort_order
     ) t) AS participants,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT pe.title AS title, pe.description AS description
         FROM project_experts pe
         WHERE pe.project_id = p.id
         ORDER BY pe.sort_order
     ) t) AS experts,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT pp.title AS title, pp.description AS description
         FROM project_partners pp
         WHERE pp.project_id = p.id
         ORDER BY pp.sort_order
     ) t) AS partners,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT pw.title AS title, pw.description AS description, pw.photo_path AS photo
         FROM project_work_formats pw
         WHERE pw.project_id = p.id
         ORDER BY pw.sort_order
     ) t) AS work_formats,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT ps.stage_number AS number, ps.description AS description
         FROM project_stages ps
         WHERE ps.project_id = p.id
         ORDER BY ps.stage_number
     ) t) AS stages,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT photo.path AS path, photo.alt AS alt, NULL::varchar AS caption
         FROM (VALUES
             (p.photo_path, p.photo_alt),
             (p.photo_2_path, p.photo_2_alt),
             (p.photo_3_path, p.photo_3_alt)
         ) AS photo(path, alt)
         WHERE photo.path IS NOT NULL
     ) t) AS photos
FROM projects p
LEFT JOIN project_details pd ON pd.project_id = p.id
WHERE p.is_active = true;

-- Статистика по проектам
CREATE OR REPLACE VIEW v_projects_stats AS
SELECT
    p.id, p.title, p.project_date,
    p.seasons_count,
    p.participants_count,
    p.experts_count,
    p.partners_count,
    (SELECT COUNT(*) FROM project_work_formats pw WHERE pw.project_id = p.id) AS work_formats_count,
    (SELECT COUNT(*) FROM project_stages ps WHERE ps.project_id = p.id) AS stages_count,
    (CASE WHEN p.photo_path IS NOT NULL THEN 1 ELSE 0 END +
     CASE WHEN p.photo_2_path IS NOT NULL THEN 1 ELSE 0 END +
     CASE WHEN p.photo_3_path IS NOT NULL THEN 1 ELSE 0 END) AS photos_count
FROM projects p WHERE p.is_active = true ORDER BY p.project_date DESC;

-- Полная карточка специалиста
CREATE OR REPLACE VIEW v_specialist_full AS
SELECT
    s.id, s.full_name, s.last_name, s.first_name, s.middle_name, s.position,
    s.short_description, s.description, s.operations_result,
    s.experience_since, s.experience_years,
    s.operations_count, s.positive_reviews_percent,
    s.photo_path, s.photo_alt, s.slug, s.is_featured, s.meta_title, s.meta_description,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT se.title AS title, se.description AS description
         FROM specialist_experience se
         WHERE se.specialist_id = s.id AND se.is_active = true
         ORDER BY se.sort_order
     ) t) AS experience,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT so.title AS title, so.description AS description
         FROM specialist_operations so
         WHERE so.specialist_id = s.id AND so.is_active = true
         ORDER BY so.sort_order
     ) t) AS operations,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT spr.title AS title, spr.description AS description
         FROM specialist_positive_reviews spr
         WHERE spr.specialist_id = s.id AND spr.is_active = true
         ORDER BY spr.sort_order
     ) t) AS positive_reviews,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT sv.id, sv.service_id, sv.is_primary, sv.sort_order,
                sv2.name AS name, sv2.slug AS slug, sub.photo_path AS photo_path
         FROM specialist_services sv
         JOIN services sv2 ON sv2.id = sv.service_id
         LEFT JOIN service_subsections sub ON sub.id = sv2.subsection_id
         WHERE sv.specialist_id = s.id AND sv.is_active = true
         ORDER BY sv.sort_order, sv.id
     ) t) AS services,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT se.education_level AS level, se.organization AS organization,
                se.year_issued AS year, se.qualification AS qualification
         FROM specialist_education se
         WHERE se.specialist_id = s.id AND se.is_active = true
         ORDER BY se.sort_order
     ) t) AS education,
    (SELECT JSON_AGG(row_to_json(t))
     FROM (
         SELECT sa.document_type AS type, sa.specialty AS specialty,
                sa.position_title AS position, sa.issue_date AS issue_date,
                sa.expiry_date AS expiry_date, sa.note AS note
         FROM specialist_accreditations sa
         WHERE sa.specialist_id = s.id AND sa.is_active = true
         ORDER BY sa.expiry_date DESC
     ) t) AS accreditations,
    (SELECT COUNT(*) FROM specialist_positive_reviews spr WHERE spr.specialist_id = s.id AND spr.is_active = true) AS reviews_count,
    (SELECT MIN(sa.expiry_date) FROM specialist_accreditations sa WHERE sa.specialist_id = s.id AND sa.is_active = true) AS next_accreditation_expiry
FROM specialists s WHERE s.is_active = true ORDER BY s.sort_order, s.last_name;

    s.id, s.full_name, s.position, s.short_description, s.photo_path, s.photo_alt, s.slug,
    s.experience_since, s.experience_years,

-- Истекающая аккредитация
CREATE OR REPLACE VIEW v_specialists_expiring_accreditations AS
SELECT
    s.id, s.full_name, s.position, sa.specialty, sa.expiry_date, sa.note,
    (sa.expiry_date - CURRENT_DATE)::integer AS days_until_expiry
FROM specialists s
JOIN specialist_accreditations sa ON s.id = sa.specialist_id
WHERE s.is_active = true AND sa.is_active = true AND sa.expiry_date <= CURRENT_DATE + INTERVAL '180 days'
ORDER BY sa.expiry_date ASC;
